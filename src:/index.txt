<canvas id="wheel"></canvas>

<script type="module">
 import { blackWheelV1 } from "./data/blackWheelV1.js";   
const canvas = document.getElementById("wheel");
const ctx = canvas.getContext("2d");

function resize() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
}

window.addEventListener("resize", resize);
resize();

const stars = blackWheelV1.stars;


let rotation = 0;

function draw() {
rotation += 0.001;

    ctx.fillStyle = "black";
    ctx.fillRect(0,0,canvas.width,canvas.height);

    const cx = canvas.width / 2;
    const cy = canvas.height / 2;

    ctx.fillStyle = "white";

   

    // Draw stars

    const wheelRadius =
    Math.min(canvas.width, canvas.height) * 0.46;

const cosR = Math.cos(rotation);
const sinR = Math.sin(rotation);

for (const star of stars) {

    const rotatedX =
        star.x * cosR -
        star.y * sinR;

    const rotatedY =
        star.x * sinR +
        star.y * cosR;

    const x = cx + rotatedX * wheelRadius;
    const y = cy + rotatedY * wheelRadius;

    ctx.beginPath();
    ctx.arc(x, y, 2, 0, Math.PI * 2);
    ctx.fill();
}
    requestAnimationFrame(draw);

}

draw();

</script>
